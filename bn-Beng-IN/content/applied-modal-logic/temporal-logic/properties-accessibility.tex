\olsection{কালগত ফ্রেমের বৈশিষ্ট্য}

আমাদের কালগত মডেলে সম্পর্ক~$\prec$-এর ওপর কোনো শর্ত আরোপ করা হয়নি। তাই পরিচিত স্বতঃসিদ্ধগুলির মধ্যে সব মডেলে কেবল~$K$ বা তার অনুরূপ $K_{\Gtemp}$ এবং~$K_{\Htemp}$ সত্য:
\begin{align*}
\tag{$K_{\Gtemp}$}  \Gtemp (p \to q) & \to (\Gtemp p \to \Gtemp q)\\
\tag{$K_{\Htemp}$}  \Htemp (p \to q) & \to (\Htemp p \to \Htemp q)
\end{align*}

কিন্তু সময়কে কীভাবে উপস্থাপন করা হবে, সে বিষয়ে আমরা মডেলের ওপর আরও কঠোর শর্ত আরোপ করতে চাইলে—যেমন $\prec$-কে রৈখিক ক্রম হতে বাধ্য করলে—আমাদের মডেলে আরও কিছু সূত্র বৈধ হবে।

\begin{table}[t]
  \begin{tabular}{| p{.48\textwidth} || p{.48\textwidth} |}
    \hline
    {\emph{$\prec$ যদি \dots হয়}} & {\emph{তবে \dots $\mModel{M}$-এ সত্য:}} \\
    \hline \hline
    \emph{সঞ্চারী}: \newline
    $\forall u \forall v \forall w ((u \prec v \land v \prec w) \lif u \prec w)$ &
    $\Ftemp \Ftemp p \lif \Ftemp p$  \\
    \hline
    \emph{রৈখিক}: \newline
    $\forall w \forall v (w \prec v \lor w = v \lor v \prec w)$ &
    $(\Ftemp \Ptemp  p \lor \Ptemp \Ftemp p) \lif (\Ptemp p \lor p \lor \Ftemp p)$\\
    \hline
    \emph{ঘন}: \newline
    $\forall w \forall v (w \prec v \to \exists u(w \prec u \land u \prec v))$ &
    $\Ftemp p \lif \Ftemp \Ftemp p$ \\
    \hline
    \emph{অতীতদিকে সীমাহীন}: \newline
    $\forall w \exists v( v \prec w)$ &
    $\Htemp p \to \Ptemp p$ \\
    \hline
    \emph{ভবিষ্যৎদিকে সীমাহীন}: \newline
    $\forall w \exists v( w \prec v)$ &
    $\Gtemp p \to \Ftemp p$ \\
    \hline
  \end{tabular}
  \caption{কালগত ফ্রেমের কয়েকটি শর্ত ও তাদের অনুরূপ সূত্র।}
  \ollabel{tab:correspondence}
\end{table}

\olref{tab:correspondence}-এ দেওয়া কয়েকটি বৈশিষ্ট্য সময়ের প্রতিনিধিত্বকারী মডেলের জন্য কাম্য মনে হতে পারে। তবে মনে রাখতে হবে, $\prec$-সম্পর্কের ওপর আমরা যে শর্তই চাই আরোপ করতে পারি বটে, কিন্তু সব শর্তের সঙ্গে কালগত যুক্তিবিদ্যার ভাষায় প্রকাশযোগ্য কোনো !!{formula}s মেলে না। যেমন, আত্মসম্পর্কহীনতা, অর্থাৎ $\forall w \lnot (w \prec w)$—এর সঙ্গে কালগত যুক্তিবিদ্যার কোনো সূত্র মেলে না।
