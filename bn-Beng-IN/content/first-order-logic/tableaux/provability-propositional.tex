\olsection{\usetoken{S}{derivability} ও বচনসংযোজক}

\begin{explain}
  ট্যাবলোর !!{derivability}-সম্পর্ক~$\Proves$ যে বচনসংযোজক-সংক্রান্ত
  কয়েকটি মৌলিক তথ্য—যেমন $!A \land !B \Proves !A$ এবং
  $!A, !A \lif !B \Proves !B$ (মোডাস পোনেন্স)—প্রতিষ্ঠা করার মতো
  যথেষ্ট শক্তিশালী, তা দেখাই। পূর্ণতা উপপাদ্যের প্রমাণে এই তথ্যগুলি
  দরকার।
\end{explain}

\begin{prop}\ollabel{prop:provability-land}
  \begin{enumerate}
  \item \ollabel{prop:provability-land-left} $!A \land !B \Proves !A$
    এবং $!A \land !B \Proves !B$ উভয়ই।
  \item \ollabel{prop:provability-land-right}
    $!A, !B \Proves !A \land !B$।
  \end{enumerate}
\end{prop}

\begin{proof}
  \begin{enumerate}
  \item $\{\sFmla{\False}{!A}, \sFmla{\True}{!A \land !B}\}$ এবং
    $\{\sFmla{\False}{!B}, \sFmla{\True}{!A \land !B}\}$ উভয়েরই
    বদ্ধ !!{tableau}s আছে:
    \begin{oltableau}{}
      [\sFmla{\False}{\formula{A}}, just=\TAss
        [\sFmla{\True}{\formula{A} \land \formula{B}}, just=\TAss
          [\sFmla{\True}{\formula{A}},just={\TRule{\True}{\land}[2]}
            [\sFmla{\True}{\formula{B}},just={\TRule{\True}{\land}[2]}, close
            ]
          ]
        ]
      ]
    \end{oltableau}
    \begin{oltableau}{}
      [\sFmla{\False}{\formula{B}}, just=\TAss
        [\sFmla{\True}{\formula{A} \land \formula{B}}, just=\TAss
          [\sFmla{\True}{\formula{A}},just={\TRule{\True}{\land}[2]}
            [\sFmla{\True}{\formula{B}},just={\TRule{\True}{\land}[2]}, close
            ]
          ]
        ]
      ]
    \end{oltableau}
    \item নিচে $\{\sFmla{\True}{!A},
      \sFmla{\True}{!B}, \sFmla{\False}{!A \land !B}\}$-এর একটি
      বদ্ধ !!{tableau} দেওয়া হলো:
      \begin{oltableau}
        [\sFmla{\False}{\formula{A} \land \formula{B}}, just = \TAss
          [\sFmla{\True}{\formula{A}}, just = \TAss
            [\sFmla{\True}{\formula{B}}, just=\TAss
              [\sFmla{\False}{\formula{A}}, just = {\TRule{\False}{\land}[1]}, close]
              [\sFmla{\False}{\formula{B}}, just = {\TRule{\False}{\land}[1]}, close]
            ]
          ]
        ]
      \end{oltableau}
  \end{enumerate}
% BN-SRC-053 TARGET-CORRECTION-BEGIN
\par\smallskip\noindent\textnormal{\small\textbf{উৎস-সংশোধন (BN-SRC-053):} হিমায়িত ইংরেজি দুই সংযোজন-বৃক্ষ ও দুই বিয়োজন-বৃক্ষে মোট আটটি \texttt{\string\sFmla} আহ্বানে সত্যমানের আর্গুমেন্ট ভুল করে \texttt{\string\True\{\ldots\}} বা \texttt{\string\False\{\ldots\}} রূপে লেখা হয়েছে, ফলে সূত্রের আলাদা দ্বিতীয় আর্গুমেন্ট নেই। বাংলা লক্ষ্যে প্রত্যেকটিকে \texttt{\string\sFmla\{\string\True\}\{\ldots\}} বা \texttt{\string\sFmla\{\string\False\}\{\ldots\}} করা হয়েছে।} % BN-SRC-053 TARGET-CORRECTION-END
\end{proof}

\begin{prop}\ollabel{prop:provability-lor}
  \begin{enumerate}
  \item $\{!A \lor !B, \lnot !A, \lnot !B\}$ অসঙ্গত।
  \item $!A \Proves !A \lor !B$ এবং $!B \Proves !A \lor !B$ উভয়ই।
  \end{enumerate}
\end{prop}

\begin{proof}
  \begin{enumerate}
  \item $\{\sFmla{\True}{!A \lor !B},
    \sFmla{\True}{\lnot !A}, \sFmla{\True}{\lnot !B}\}$-এর একটি
    বদ্ধ !!{tableau} দিই:
    \begin{oltableau}
      [\sFmla{\True}{\formula{A} \lor \formula{B}}, just = \TAss
        [\sFmla{\True}{\lnot \formula{A}}, just = \TAss
          [\sFmla{\True}{\lnot \formula{B}}, just = \TAss
            [\sFmla{\False}{\formula{A}}, just = {\TRule{\True}{\lnot}[2]}
              [\sFmla{\False}{\formula{B}}, just = {\TRule{\True}{\lnot}[3]}
                [\sFmla{\True}{\formula{A}}, just = {\TRule{\True}{\lor}[1]}, close]
                [\sFmla{\True}{\formula{B}}, just = {\TRule{\True}{\lor}[1]}, close]
              ]
            ]
          ]
        ]
      ]
    \end{oltableau}
  \item $\{\sFmla{\False}{!A \lor !B}, \sFmla{\True}{!A}\}$ এবং
    $\{\sFmla{\False}{!A \lor !B}, \sFmla{\True}{!B}\}$ উভয়েরই
    বদ্ধ !!{tableau}s আছে:
    \begin{oltableau}{}
      [\sFmla{\False}{\formula{A} \lor \formula{B}}, just=\TAss
        [\sFmla{\True}{\formula{A}}, just=\TAss
          [\sFmla{\False}{\formula{A}},just={\TRule{\False}{\lor}[1]}
            [\sFmla{\False}{\formula{B}},just={\TRule{\False}{\lor}[1]}, close
            ]
          ]
        ]
      ]
    \end{oltableau}
    \begin{oltableau}{}
      [\sFmla{\False}{\formula{A} \lor \formula{B}}, just=\TAss
        [\sFmla{\True}{\formula{B}}, just=\TAss
          [\sFmla{\False}{\formula{A}},just={\TRule{\False}{\lor}[1]}
            [\sFmla{\False}{\formula{B}},just={\TRule{\False}{\lor}[1]}, close
            ]
          ]
        ]
      ]
    \end{oltableau}
  \end{enumerate}
\end{proof}

\begin{prop}\ollabel{prop:provability-lif}
  \begin{enumerate}
  \item \ollabel{prop:provability-lif-left}
    $!A, !A \lif !B \Proves !B$।
  \item \ollabel{prop:provability-lif-right}
    $\lnot !A \Proves !A \lif !B$ এবং $!B \Proves !A \lif !B$
    উভয়ই।
  \end{enumerate}
\end{prop}

\begin{proof}
  \begin{enumerate}
    \item $\{\sFmla{\False}{!B}, \sFmla{\True}{!A \lif !B},
      \sFmla{\True}{!A}\}$-এর একটি বদ্ধ !!{tableau} আছে:
      \begin{oltableau}
        [\sFmla{\False}{\formula{B}}, just=\TAss
          [\sFmla{\True}{\formula{A} \lif \formula{B}}, just=\TAss
            [\sFmla{\True}{\formula{A}}, just=\TAss
              [\sFmla{\False}{\formula{A}}, just = {\TRule{\True}{\lif}[2]}, close]
              [\sFmla{\True}{\formula{B}}, just = {\TRule{\True}{\lif}[2]}, close]
            ]
          ]
        ]
      \end{oltableau}
    \item $\{\sFmla{\False}{!A \lif !B},
      \sFmla{\True}{\lnot !A}\}$ এবং $\{\sFmla{\False}{!A \lif !B},
      \sFmla{\True}{!B}\}$ উভয়েরই বদ্ধ !!{tableau}s আছে:
      \begin{oltableau}
        [\sFmla{\False}{\formula{A} \lif \formula{B}}, just = \TAss
          [\sFmla{\True}{\lnot \formula{A}}, just = \TAss
            [\sFmla{\True}{\formula{A}}, just = {\TRule{\False}{\lif}[1]}
              [\sFmla{\False}{\formula{B}}, just = {\TRule{\False}{\lif}[1]}
                [\sFmla{\False}{\formula{A}}, just = {\TRule{\True}{\lnot}[2]}, close]
              ]
            ]
          ]
        ]
      \end{oltableau}
      \begin{oltableau}
        [\sFmla{\False}{\formula{A} \lif \formula{B}}, just = \TAss
          [\sFmla{\True}{\formula{B}}, just = \TAss
            [\sFmla{\True}{\formula{A}}, just = {\TRule{\False}{\lif}[1]}
              [\sFmla{\False}{\formula{B}}, just = {\TRule{\False}{\lif}[1]},close
              ]
            ]
          ]
        ]
      \end{oltableau}
  \end{enumerate}
\end{proof}
