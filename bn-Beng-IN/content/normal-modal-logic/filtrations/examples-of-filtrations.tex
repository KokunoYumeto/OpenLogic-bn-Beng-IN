\olsection{ফিল্ট্রেশনের উদাহরণ}

এখনও দেখাইনি যে কোনো ফিল্ট্রেশন আছে। বাস্তবে যেকোনো মডেল $\mModel{M}$-এর $\Gamma$-র মধ্য দিয়ে বহু ফিল্ট্রেশন আছে। বিশেষ করে দুটি চিহ্নিত করব: সবচেয়ে সূক্ষ্ম ও সবচেয়ে স্থূল ফিল্ট্রেশন। একই মডেল $\mModel{M}$-এর ফিল্ট্রেশনগুলির পার্থক্য তাদের অভিগম্যতা-সম্বন্ধে (\olref[fil]{defn:filtration}-তে $W^*$ ও~$V^*$ সরাসরি নির্দিষ্ট)। সবচেয়ে সূক্ষ্ম ফিল্ট্রেশনে সম্পর্কিত জগৎ-জোড়া যথাসম্ভব কম, আর সবচেয়ে স্থূল ফিল্ট্রেশনে যথাসম্ভব বেশি।

\begin{defn}
  $\Gamma$ উপসূত্রের অধীনে বদ্ধ হলে, মডেল $\mModel{M}$-এর \emph{সবচেয়ে সূক্ষ্ম} ফিল্ট্রেশন $\mModel{M^*}$ এভাবে সংজ্ঞায়িত:
  \[
  R^*[u][v] \quad \text{তখনই এবং কেবল তখনই, যখন} \quad \exists u'\in [u] \;
  \exists v' \in [v] : Ru'v'.
  \]
\end{defn}

\begin{prop}\ollabel{prop:finest}
  সবচেয়ে সূক্ষ্ম ফিল্ট্রেশন $\mModel{M^*}$ বাস্তবেই একটি ফিল্ট্রেশন।
\end{prop}

\begin{proof}
  এভাবে সংজ্ঞায়িত $R^*$ যে \olref[fil]{defn:filtration}\olref[fil]{defn:filtration-R} মেনে চলে, তা পরীক্ষা করতে হবে। তিনটি শর্ত পরপর দেখি।

  $Ruv$ হলে $u \in [u]$ ও $v \in [v]$ বলে $R^*[u][v]$-ও; তাই \olref[fil]{defn:filtration-R1} পূরণ হয়।

  \iftag{prvBox}{\iftag{probBox}{\olref[fil]{defn:filtration-R2}-এর যাচাই অনুশীলনী হিসাবে থাক।}{\olref[fil]{defn:filtration-R2}-এর জন্য ধরি $\Box!A \in \Gamma$, $R^*[u][v]$ এবং $\mSat{M}{\Box!A}[u]$। $R^*$-এর সংজ্ঞায় এমন $u' \equiv u$ ও $v' \equiv v$ আছে যাতে $Ru'v'$। $u$ ও $u'$ $\Gamma$-তে একমত বলে $\mSat{M}{\Box!A}[u']$-ও; ফলে $\mSat{M}{!A}[v']$। $\Gamma$ উপ-!!{formula}s-এর অধীনে বদ্ধ বলে $v$ ও $v'$ $!A$-তে একমত; অতএব কাঙ্ক্ষিত $\mSat{M}{!A}[v]$।}}{}

  \iftag{prvDiamond}{\iftag{probDiamond}{\olref[fil]{defn:filtration-R3}-এর যাচাই অনুশীলনী হিসাবে থাক।}{\olref[fil]{defn:filtration-R3} যাচাই করতে ধরি $\Diamond!A \in
        \Gamma$, $R^*[u][v]$ এবং $\mSat{M}{!A}[v]$। $R^*$-এর সংজ্ঞায় এমন $u' \equiv u$ ও $v' \equiv v$ আছে যাতে $Ru'v'$। $v$ ও $v'$ $\Gamma$-তে একমত এবং $\Gamma$ উপ-!!{formula}s-এর অধীনে বদ্ধ বলে $\mSat{M}{!A}[v']$-ও; ফলে $\mSat{M}{\Diamond !A}[u']$। $u$ ও $u'$-ও $\Gamma$-তে একমত বলে $\mSat{M}{\Diamond !A}[u]$।}}{}
\end{proof}

\begin{probtag}{probBox,probDiamond}
  \olref[nml][fil][exf]{prop:finest}-এর প্রমাণ সম্পূর্ণ করো।
\end{probtag}

\begin{defn}
  $\Gamma$ উপসূত্রের অধীনে বদ্ধ হলে, মডেল~$\mModel{M}$-এর \emph{সবচেয়ে স্থূল} ফিল্ট্রেশন~$\mModel{M^*}$-এ $R^*[u][v]$ তখনই এবং কেবল তখনই, যখন \iftag{notprvBox,notprvDiamond}{নিচের শর্তটি পূরণ হয়:}{নিচের \emph{উভয়} শর্ত পূরণ হয়:}
  \begin{tagenumerate}{prvBox,prvDiamond}
  \tagitem{prvBox}{\ollabel{defn:coarsest-Box}$\Box!A \in \Gamma$
    এবং $\mSat{M}{\Box!A}[u]$ হলে
    $\mSat{M}{!A}[v]$\iftag{prvDiamond}{;}{.}}{}
  \tagitem{prvDiamond}{\ollabel{defn:coarsest-Diamond}$\Diamond!A
    \in \Gamma$ এবং $\mSat{M}{!A}[v]$ হলে
    $\mSat{M}{\Diamond!A}[u]$.}{}
  \end{tagenumerate}
\end{defn}

\begin{prop}
  সবচেয়ে স্থূল ফিল্ট্রেশন $\mModel{M^*}$ বাস্তবেই একটি ফিল্ট্রেশন।
\end{prop}

\begin{proof}
  $R^*$-এর সংজ্ঞা অনুযায়ী শুধু $Ruv$ থেকে $R^*[u][v]$-র অনুসৃতি যাচাই করা বাকি। তাই ধরি $Ruv$। \iftag{prvBox}{ধরি $\Box!A \in \Gamma$ এবং $\mSat{M}{\Box!A}[u]$; স্পষ্টত $\mSat{M}{!A}[v]$\iftag{prvDiamond}{, তাই \olref{defn:coarsest-Box} পূরণ হয়। }{।}}{}\iftag{prvDiamond}{ধরি $\Diamond!A \in
    \Gamma$ এবং $\mSat{M}{!A}[v]$। $Ruv$ বলে $\mSat{M}{\Diamond!A}[u]$\iftag{prvBox}{, তাই \olref{defn:coarsest-Diamond} পূরণ হয়}{}।}{}
\end{proof}

\begin{ex}
  ধরা যাক $W = \PosInt$, $Rnm$ তখনই, যখন $m = n + 1$, এবং $V(p) = \Setabs{2n}{n \in
    \Nat}$। \olref{fig:ex-filtration}-এ মডেল $\mModel{M} = \tuple{W, R, V}$ দেখানো হয়েছে। জগৎগুলি $1$, $2$ ইত্যাদি; প্রতিটি জগৎ ঠিক একটি অন্য জগতে—তার পরবর্তী জগতে—অভিগম্য, এবং $p$ কেবল জোড় সংখ্যাগুলিতেই সত্য।

  এবার $\Gamma$ ধরি~$\Box p \lif p$-র উপ-!!{formula}s-এর সমষ্টি, অর্থাৎ $\{p, \Box p, \Box p \lif p\}$। $p$ কেবল জোড় সংখ্যায় সত্য, $\Box p$ কেবল বিজোড় সংখ্যায় সত্য, তাই $\Box p \lif p$ কেবল জোড় সংখ্যায় সত্য। অর্থাৎ প্রত্যেক বিজোড় সংখ্যায় $\Box p$ সত্য, কিন্তু $p$ ও $\Box
  p \lif p$ মিথ্যা; প্রত্যেক জোড় সংখ্যায় $p$ ও $\Box p \lif p$ সত্য, কিন্তু $\Box p$ মিথ্যা। তাই $W^* = \{ [1], [2] \}$, যেখানে $[1] =
  \{1, 3, 5, \dots\}$ এবং $[2] = \{2, 4, 6, \dots\}$। যেহেতু $2 \in
  V(p)$, তাই $[2] \in V^*(p)$; যেহেতু $1 \notin V(p)$, তাই $[1] \notin
  V^*(p)$। সুতরাং $V^*(p) = \{[2]\}$।

  $W^*$-ভিত্তিক যেকোনো ফিল্ট্রেশনের অভিগম্যতা-সম্বন্ধে $\tuple{[1], [2]}, \tuple{[2],[1]}$ থাকতে হবে: $R12$ বলে \olref[fil]{defn:filtration}\olref[fil]{defn:filtration-R1} থেকে $R^*[1][2]$; আর $R23$ বলে $R^*[2][3]$, যেখানে $[3]=[1]$। এতে $\tuple{[1],[1]}$ থাকতে পারে না: থাকলে $R^*[1][1]$ হতো, অথচ $\mSat{M}{\Box p}[1]$ এবং $\mSat/{M}{p}[1]$; এটি \olref[fil]{defn:filtration-R2}-এর বিরোধী। $R^*[2][2]$ থাকা বাধ্যতামূলকও নয়, নিষিদ্ধও নয়। তাই~$\mModel{M}$-এর সম্ভাব্য ফিল্ট্রেশন দুটি; তাদের অভিগম্যতা-সম্বন্ধ যথাক্রমে
  \[
  \{\tuple{[1],[2]}, \tuple{[2],[1]}\} \text{ and }
  \{\tuple{[1],[2]}, \tuple{[2],[1]}, \tuple{[2],[2]}\}.
  \]
  উভয় ক্ষেত্রেই~$[1]$-তে $p$ ও $\Box p \lif p$ মিথ্যা এবং $\Box p$ সত্য; $[2]$-তে $p$ ও $\Box p \lif p$ সত্য এবং $\Box p$ মিথ্যা।
\end{ex}
