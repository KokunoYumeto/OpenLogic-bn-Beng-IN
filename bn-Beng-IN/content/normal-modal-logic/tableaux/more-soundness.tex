\olsection{অতিরিক্ত বিধিগুলির বিশুদ্ধতা}

কোনো মডেল-শ্রেণির জন্য একটি বিধিকে বিশুদ্ধ বলি, যদি !!a{tableau}-র
কোনো শাখা ওই শ্রেণির একটি মডেলে পরিতৃপ্তিযোগ্য হলে, বিধি প্রয়োগে
পাওয়া শাখাটিও ওই শ্রেণির একটি মডেলে পরিতৃপ্তিযোগ্য হয়।

\begin{prop}
  \ollabel{prop:soundness-T}
  \iftag{prvBox}{\Ax{T}$\Box$}{}%
  \iftag{notprvBox,notprvDiamond}{}{ ও }%
  \iftag{prvDiamond}{\Ax{T}$\Diamond$}{}
  \iftag{notprvBox,notprvDiamond}{বিধিটি}{বিধিগুলি}
  প্রতিবিম্বী মডেলের জন্য বিশুদ্ধ।
\end{prop}

\begin{proof}
\begin{tagenumerate}{prvBox,prvDiamond}
\tagitem{prvBox}{$\sFmla{\True}{\Box !B}[\sigma] \in \Gamma$-তে
  T$\Box$ প্রয়োগে শাখাটি প্রসারিত হয়:
  \iftag{probBox}{অনুশীলন।}{শাখায় নতুন !!{signed
      formula}~$\sFmla{\True}{!B}[\sigma]$ পাওয়া যায়। ধরি
    $\mSat{M}{\Gamma}[f]$, বিশেষত
    $\mSat{M}{\Box !B}[f(\sigma)]$। $R$ প্রতিবিম্বী হওয়ায়
    $Rf(\sigma)f(\sigma)$। তাই
    $\mSat{M}{!B}[f(\sigma)]$; অর্থাৎ
    $\mModel{M}, f$, $\sFmla{\True}{!B}[\sigma]$
    পরিতৃপ্ত করে।}}{}
\tagitem{prvDiamond}{$\sFmla{\False}{\Diamond !B}[\sigma] \in \Gamma$-তে
  T$\Diamond$ প্রয়োগে শাখাটি প্রসারিত হয়:
  \iftag{probDiamond}{অনুশীলন।}{শাখায় নতুন !!{signed
      formula}~$\sFmla{\False}{!B}[\sigma]$ পাওয়া যায়। ধরি
    $\mSat{M}{\Gamma}[f]$, বিশেষত
    $\mSat/{M}{\Diamond !B}[f(\sigma)]$। $R$ প্রতিবিম্বী হওয়ায়
    $Rf(\sigma)f(\sigma)$। তাই
    $\mSat/{M}{!B}[f(\sigma)]$; অর্থাৎ
    $\mModel{M}, f$, $\sFmla{\False}{!B}[\sigma]$
    পরিতৃপ্ত করে।}}{}
\end{tagenumerate}
\end{proof}

\tagprob{probBox,probDiamond}
\begin{prob}
  \olref[nml][tab][msn]{prop:soundness-T}-এর প্রমাণ সম্পূর্ণ করো।
\end{prob}
\tagendprob

\begin{prop}

  প্রতিসম মডেলের জন্য বিশুদ্ধ।
\end{prop}

\begin{proof}
\begin{tagenumerate}{prvBox,prvDiamond}
\tagitem{prvBox}{$\sFmla{\True}{\Box !B}[\sigma.n] \in \Gamma$-তে
  B$\Box$ প্রয়োগে শাখাটি প্রসারিত হয়:

      formula}~$\sFmla{\True}{!B}[\sigma]$ পাওয়া যায়।

    $\mSat{M}{\Box !B}[f(\sigma.n)]$। $f$ শাখার
    পূর্বসূচিগুলির $\mModel{M}$-এ ব্যাখ্যা হওয়ায়
    $Rf(\sigma)f(\sigma.n)$। $R$ প্রতিসম, তাই
    $Rf(\sigma.n)f(\sigma)$। যেহেতু
    $\mSat{M}{\Box !B}[f(\sigma.n)]$,
    তাই $\mSat{M}{!B}[f(\sigma)]$। অতএব

    ধরি $\mSat{M}{\Gamma}[f]$, বিশেষত
    $\mSat/{M}{\Diamond !B}[f(\sigma.n)]$। $f$ শাখার
    পূর্বসূচিগুলির $\mModel{M}$-এ ব্যাখ্যা হওয়ায়
